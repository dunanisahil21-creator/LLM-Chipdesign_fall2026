module mux32_1(input [4:0] sel, input [31:0] in, output reg [31:0] out);
    wire [15:0] mux0_out, mux1_out;

    mux16_1 mux0 (
        .sel(sel[3:0]),
        .in(in[15:0]),
        .out(mux0_out)
    );

    mux16_1 mux1 (
        .sel(sel[3:0]),
        .in(in[31:16]),
        .out(mux1_out)
    );

    always @(*) begin
        case(sel[4])
            1'b0: out = {16'b0, mux0_out}; 
            1'b1: out = {16'b0, mux1_out}; 
        endcase
    end
endmodule
